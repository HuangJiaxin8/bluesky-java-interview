public class CacheUtil<T,R> {  
  
    private Map<T, R> cacheMap = new ConcurrentHashMap<>();  
  
    public List<R> getCache(FunctionPlus<T, R> fun, List<T> ids){  
        List<R> result = new ArrayList<R>();  
        List<T> cacheIds = new ArrayList<>();  
  
        for (T id : ids) {  
            if (cacheMap.containsKey(id)) {  
                cacheIds.add(id);  
                result.add(cacheMap.get(id));  
            }  
        }        
        ids.removeAll(cacheIds);  
  
        Map<T, R> apply = fun.apply(ids);  
        cacheMap.putAll(apply);  
  
        // 更新缓存  
        List<R> infoList = apply.values().stream().collect(Collectors.toList());  
        result.addAll(infoList);  
        return result;  
    }  
}



@FunctionalInterface  
public interface FunctionPlus<T, R> {  
    Map<T, R> apply(List<T> t);  
}
